	\vspace{0.3cm}
	
	\section*{1. Solución General}
	\[
	y = y_c + y_p
	\]
	\begin{itemize}
		\item $y_c$: Solución homogénea ($R(x) = 0$).
		\item $y_p$: Solución particular debida a $R(x)$.
	\end{itemize}
	
	\hrulefill
	
	\section*{2. Solución Complementaria $y_c$}
	Se sustituyen las derivadas por potencias de $m$ ($y'' \to m^2$, $y' \to m$, $y \to 1$) e igualamos a 0.
	
	\begin{itemize}
		\item \textbf{Caso 1: Raíces reales y distintas ($m_1 \neq m_2 \neq \dots \neq m_n$)}
		\[
		y_c = C_1 e^{m_1 x} + C_2 e^{m_2 x} + \dots + C_n e^{m_n x}
		\]
		
		\item \textbf{Caso 2: Raíces reales repetidas ($m_1 = m_2 = \dots = m$)}
		\[
		y_c = C_1 e^{m x} + C_2 x e^{m x} + C_3 x^2 e^{m x} + \dots + C_n x^{n-1} e^{m x}
		\]
		
		\item \textbf{Caso 3: Raíces complejas conjugadas ($m = \alpha \pm \beta i$)}
		\[
		y_c = e^{\alpha x} \left( C_1 \cos(\beta x) + C_2 \sin(\beta x) \right)
		\]
		\textit{Si $\alpha = 0$ ($m = \pm \beta i$): $y_c = C_1 \cos(\beta x) + C_2 \sin(\beta x)$.}
		
		\item \textbf{Caso 4: Raíces complejas conjugadas repetidas ($k$ veces)}
		\[
		y_c = e^{\alpha x} \left[ (C_1 + C_2 x + \dots)\cos(\beta x) + (C_3 + C_4 x + \dots)\sin(\beta x) \right]
		\]
	\end{itemize}
